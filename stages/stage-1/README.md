## Apply
```powershell
$e='598fd1b0-36f1-402f-ba36-aa00c8a67cc4'; $sk='cf00d339-5217-4cd1-b390-cc0d911735da'
$raw = frontier-tuning skills get $sk --env-id $e -o json | Out-String; $s = $raw.Substring($raw.IndexOf('{')) | ConvertFrom-Json
$s.Rubrics = Get-Content stages\stage-1\warranty-assistant.rubrics.json -Raw | ConvertFrom-Json
$s | ConvertTo-Json -Depth 12 | Set-Content "$env:TEMP\payload.json" -Encoding utf8
frontier-tuning skills update $sk --file "$env:TEMP\payload.json" --env-id $e
# replace the 30 Evaluation samples so they carry the new rubrics
frontier-tuning samples list --limit 500 --env-id $e -o json        # then: samples delete <id> --yes, for each
frontier-tuning samples upload stages\stage-1\samples.jsonl --skill-id $sk --type Evaluation --env-id $e
```
